    68→function neuralPath(cx: number, cy: number, tx: number, ty: number): string {
    69→  const dx = tx - cx;
    70→  const dy = ty - cy;
    71→  const cp1x = cx + dx * 0.35 + dy * 0.08;
    72→  const cp1y = cy + dy * 0.35 - dx * 0.08;
    73→  const cp2x = cx + dx * 0.65 - dy * 0.08;
    74→  const cp2y = cy + dy * 0.65 + dx * 0.08;
    75→  return `M ${cx} ${cy} C ${cp1x} ${cp1y}, ${cp2x} ${cp2y}, ${tx} ${ty}`;
    76→}
    77→
    78→function bezierPoint(
    79→  t: number,
    80→  p0: [number, number],
    81→  p1: [number, number],
    82→  p2: [number, number],
    83→  p3: [number, number]
    84→): [number, number] {
    85→  const mt = 1 - t;
    86→  const x = mt * mt * mt * p0[0] + 3 * mt * mt * t * p1[0] + 3 * mt * t * t * p2[0] + t * t * t * p3[0];
    87→  const y = mt * mt * mt * p0[1] + 3 * mt * mt * t * p1[1] + 3 * mt * t * t * p2[1] + t * t * t * p3[1];
    88→  return [x, y];
    89→}
    90→
    91→function getControlPoints(cx: number, cy: number, tx: number, ty: number): [[number, number], [number, number]] {
    92→  const dx = tx - cx;
    93→  const dy = ty - cy;
    94→  return [
    95→    [cx + dx * 0.35 + dy * 0.08, cy + dy * 0.35 - dx * 0.08],
    96→    [cx + dx * 0.65 - dy * 0.08, cy + dy * 0.65 + dx * 0.08],
    97→  ];
    98→}
