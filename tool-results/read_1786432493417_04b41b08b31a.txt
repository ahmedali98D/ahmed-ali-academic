    91→// ─── Typing Effect Quote ─────────────────────────────────
    92→function TypingQuote({ text }: { text: string }) {
    93→  const [displayed, setDisplayed] = useState('')
    94→  const [started, setStarted] = useState(false)
    95→
    96→  useEffect(() => {
    97→    const timer = setTimeout(() => setStarted(true), 2000)
    98→    return () => clearTimeout(timer)
    99→  }, [])
   100→
   101→  // Reset the typing animation whenever the text (language) changes
   102→  useEffect(() => {
   103→    setDisplayed('')
   104→  }, [text])
   105→
   106→  useEffect(() => {
   107→    if (!started) return
   108→    if (displayed.length < text.length) {
   109→      const timer = setTimeout(() => {
   110→        setDisplayed(text.slice(0, displayed.length + 1))
   111→      }, 50)
   112→      return () => clearTimeout(timer)
   113→    }
   114→  }, [displayed, started, text])
   115→
   116→  return (
   117→    <span>
   118→      {displayed}
   119→      {displayed.length < text.length && started && (
   120→        <span className="animate-typing-cursor text-amber-400">|</span>
   121→      )}
   122→    </span>
   123→  )
   124→}
