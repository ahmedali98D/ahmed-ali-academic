     1→/**
     2→ * Email sending via Brevo.
     3→ *
     4→ * PRIMARY: Brevo HTTP API (https://api.brevo.com/v3/smtp/email)
     5→ *   The API sometimes routes through different IP pools than the SMTP
     6→ *   relay, which can improve deliverability to Microsoft 365 / Outlook
     7→ *   domains that aggressively filter Brevo's shared SMTP IPs.
     8→ *
     9→ * FALLBACK: Brevo SMTP relay (smtp-relay.brevo.com:587)
    10→ *   If the API call fails (network error, 5xx response), we fall back
    11→ *   to the SMTP relay so a transient API outage doesn't block email.
    12→ *
    13→ * The API key is the same value for both paths (starts with xkeysib-).
    14→ * Reuses the existing SMTP_PASS env var — no new env vars needed.
    15→ *
    16→ * Required env vars:
    17→ *   SMTP_HOST       = smtp-relay.brevo.com
    18→ *   SMTP_PORT       = 587
    19→ *   SMTP_USER       = (your Brevo SMTP login, usually your Brevo account email)
    20→ *   SMTP_PASS       = (your Brevo key — starts with xkeysib-. Used for BOTH
    21→ *                     SMTP auth AND the API key header)
    22→ *   CONTACT_EMAIL   = ahmed@phronesis-studio.com (the From address)
    23→ *   CONTACT_NAME    = "Ahmed Ali — Studio of Phronesis"
    24→ *   NOTIFY_EMAIL    = ahmed@phronesis-studio.com (where notifications go —
    25→ *                     forwarded to Gmail via ImprovMX)
    26→ */
    27→import nodemailer from "nodemailer";
    28→import { resolveMx, resolveTxt, resolveCname } from "dns/promises";
    29→
    30→let cachedTransport: nodemailer.Transporter | null = null;
    31→
    32→function getTransport(): nodemailer.Transporter {
    33→  if (cachedTransport) return cachedTransport;
    34→
    35→  const host = process.env.SMTP_HOST;
    36→  const port = parseInt(process.env.SMTP_PORT || "587", 10);
    37→  const user = process.env.SMTP_USER;
    38→  const pass = process.env.SMTP_PASS;
    39→
    40→  if (!host || !user || !pass) {
    41→    throw new Error(
    42→      "SMTP env vars not set. Need SMTP_HOST, SMTP_USER, SMTP_PASS."
    43→    );
    44→  }
    45→
    46→  cachedTransport = nodemailer.createTransport({
    47→    host,
    48→    port,
    49→    secure: port === 465,
    50→    auth: { user, pass },
    51→  });
    52→
    53→  return cachedTransport;
    54→}

    56→function getFrom(): { name: string; address: string } {
    57→  const email = process.env.CONTACT_EMAIL || "ahmed@phronesis-studio.com";
    58→  const name = process.env.CONTACT_NAME || "Ahmed Ali · Studio of Phronesis";
    59→  return { name, address: email };
    60→}

    62→export type EmailPayload = {
    63→  to: string;
    64→  subject: string;
    65→  text?: string; // plain-text fallback
    66→  html?: string; // rich HTML body
    67→  replyTo?: string;

    70→/**
    71→ * Result of sendEmail — includes the SMTP server's response so callers can
    72→ * confirm Brevo actually accepted the message (not just that the API call
    73→ * resolved). The `messageId` is Brevo's internal tracking ID; the user can
    74→ * look it up in Brevo dashboard → Transactional → Logs to see what happened
    75→ * to the email AFTER Brevo accepted it (delivered / bounced / deferred / blocked).
    76→ */
    77→export type EmailSendResult = {
    78→  messageId: string | null;
    79→  response: string;       // e.g. "250 Ok <queued messageId>"
    80→  accepted: string[];     // recipient addresses Brevo accepted
    81→  rejected: string[];     // recipient addresses Brevo rejected
    82→  pending: string[];      // addresses pending
    83→};
    84→
    85→/**
    86→ * Send an email. PRIMARY path: Brevo HTTP API. FALLBACK: Brevo SMTP relay.
    87→ *
    88→ * The API is preferred because it sometimes routes through a different IP
    89→ * pool than the SMTP relay — this matters for deliverability to Microsoft 365
    90→ * / Outlook domains, which aggressively filter Brevo's shared SMTP IPs.
    91→ *
    92→ * If the API call fails (network error, non-2xx response), we automatically
    93→ * fall back to the SMTP relay so a transient API outage doesn't block email.
    94→ *
    95→ * Returns an EmailSendResult. Inspect `rejected` — if non-empty, the
    96→ * recipient was rejected. If `accepted` is empty, nothing was sent.
    97→ *
    98→ * IMPORTANT: resolving does NOT guarantee delivery. Brevo may still:
    99→ *   - Defer delivery if the recipient's MX is strict about IP reputation
   100→ *   - Bounce later if the recipient's mailbox is full or doesn't exist
   101→ *   - Be silently dropped by the recipient's gateway (e.g. Microsoft 365 EOP)
   102→ * The `messageId` is the only way to track what happened post-acceptance.
   103→ * Look it up in Brevo dashboard → Transactional → Logs.
   104→ */
   105→export async function sendEmail(p: EmailPayload): Promise<EmailSendResult> {
   106→  // Try the API first
   107→  try {
   108→    const apiResult = await sendEmailViaApi(p);
   109→    if (apiResult.accepted.length > 0) {
   110→      return apiResult;
   111→    }
   112→    // If accepted is empty but no exception was thrown, fall through to SMTP
   113→    console.warn("[email] Brevo API returned no accepted recipients, falling back to SMTP");
   114→  } catch (apiErr) {
   115→    console.warn("[email] Brevo API failed, falling back to SMTP:", apiErr instanceof Error ? apiErr.message : apiErr);
   116→  }
   117→
   118→  // Fallback to SMTP
   119→  return sendEmailViaSmtp(p);
   120→}
   121→
   122→/**
   123→ * Send via Brevo HTTP API (POST https://api.brevo.com/v3/smtp/email).
   124→ *
   125→ * API key resolution:
   126→ *   1. BREVO_API_KEY env var (preferred — generate in Brevo dashboard →
   127→ *      SMTP & API → API Keys → Generate)
   128→ *   2. SMTP_PASS env var (fallback — in SOME Brevo accounts the SMTP key
   129→ *      also works as an API key; if not, the API call will 401 and the
   130→ *      caller falls back to SMTP)
   131→ *
   132→ * If you're seeing SMTP-format responses ("250 2.0.0 OK: queued as...")
   133→ * in the admin diagnostic instead of "API 200 OK", it means the API key
   134→ * is not set or invalid, and the code is falling back to SMTP. Generate
   135→ * a Brevo API key and add it as BREVO_API_KEY in Vercel env vars.
   136→ */
   137→async function sendEmailViaApi(p: EmailPayload): Promise<EmailSendResult> {
   138→  const apiKey = process.env.BREVO_API_KEY || process.env.SMTP_PASS;
   139→  if (!apiKey) {
   140→    throw new Error("No Brevo API key found. Set BREVO_API_KEY (preferred) or SMTP_PASS env var.");
   141→  }
   142→
   143→  const from = getFrom();
   144→  // Brevo API expects recipients as an array of { email, name? } objects.
   145→  // The `to` field in EmailPayload may be a plain address or "Name <addr>".
   146→  const { address: toAddress, name: toName } = parseEmailAddress(p.to);
   147→
   148→  const body: Record<string, unknown> = {
   149→    sender: { name: from.name, email: from.address },
   150→    to: [{ email: toAddress, ...(toName ? { name: toName } : {}) }],
   151→    subject: p.subject,
   152→  };
   153→  if (p.html) body.htmlContent = p.html;
   154→  if (p.text) body.textContent = p.text;
   155→  if (p.replyTo) {
   156→    const { address: replyAddress, name: replyName } = parseEmailAddress(p.replyTo);
   157→    body.replyTo = { email: replyAddress, ...(replyName ? { name: replyName } : {}) };
   158→  }
   159→
   160→  const resp = await fetch("https://api.brevo.com/v3/smtp/email", {
   161→    method: "POST",
   162→    headers: {
   163→      "accept": "application/json",
   164→      "content-type": "application/json",
   165→      "api-key": apiKey,
   166→    },
   167→    body: JSON.stringify(body),
   168→  });
   169→
   170→  const respText = await resp.text();
   171→
   172→  if (!resp.ok) {
   173→    // Brevo API returns 400 for validation errors, 401 for bad API key,
   174→    // 429 for rate limit, 5xx for server errors.
   175→    throw new Error(`Brevo API HTTP ${resp.status}: ${respText.slice(0, 500)}`);
   176→  }
   177→
   178→  // Brevo API returns { "messageId": "xxx@domain.com" }
   179→  let respJson: { messageId?: string };
   180→  try {
   181→    respJson = JSON.parse(respText);
   182→  } catch {
   183→    respJson = {};
   184→  }
   185→
   186→  const messageId = respJson.messageId
   187→    ? respJson.messageId.replace(/^<|>$/g, "")
   188→    : null;
   189→
   190→  return {
   191→    messageId,
   192→    response: `API ${resp.status} OK`,
   193→    accepted: [toAddress],
   194→    rejected: [],
   195→    pending: [],
   196→  };
   197→}

   200→ * Send via Brevo SMTP relay (the original path, kept as fallback).
   201→ */
   202→async function sendEmailViaSmtp(p: EmailPayload): Promise<EmailSendResult> {
   203→  const transport = getTransport();
   204→  const info = await transport.sendMail({
   205→    from: getFrom(),
   206→    to: p.to,
   207→    subject: p.subject,
   208→    text: p.text,
   209→    html: p.html,
   210→    replyTo: p.replyTo,
   211→  });
   212→
   213→  const messageId = info.messageId
   214→    ? info.messageId.replace(/^<|>$/g, "")
   215→    : null;
   216→
   217→  const smtpInfo = info as unknown as {
   218→    accepted?: string[];
   219→    rejected?: string[];
   220→    pending?: string[];
   221→    response?: string;
   222→  };
   223→
   224→  return {
   225→    messageId,
   226→    response: smtpInfo.response || "(no SMTP response)",
   227→    accepted: smtpInfo.accepted || [],
   228→    rejected: smtpInfo.rejected || [],
   229→    pending: smtpInfo.pending || [],
   230→  };
   231→}

   233→/**
   234→ * Parse an email address that may be in either format:
   235→ *   "user@example.com"           → { address: "user@example.com", name: "" }
   236→ *   "Name <user@example.com>"    → { address: "user@example.com", name: "Name" }
   237→ */
   238→function parseEmailAddress(input: string): { address: string; name: string } {
   239→  const match = input.match(/^\s*(.*?)\s*<([^>]+)>\s*$/);
   240→  if (match) {
   241→    return { name: match[1].replace(/^["']|["']$/g, ""), address: match[2].trim() };
   242→  }
   243→  return { name: "", address: input.trim() };
   244→}

   246→/**
   247→ * Verify SMTP credentials by issuing a NOOP command. Returns ok=true if the
   248→ * SMTP server accepted the connection and authenticated successfully.
   249→ * Does NOT verify that the From address is a Brevo Verified Sender —
   250→ * that's a separate check the user must do in the Brevo dashboard.
   251→ */
   252→export async function verifySmtp(): Promise<{
   253→  ok: boolean;
   254→  message: string;
   255→}> {
   256→  try {
   257→    const transport = getTransport();
   258→    await transport.verify();
   259→    return {
   260→      ok: true,
   261→      message: `SMTP connection verified. Host: ${process.env.SMTP_HOST}, user: ${process.env.SMTP_USER}`,
   262→    };
   263→  } catch (err) {
   264→    return {
   265→      ok: false,
   266→      message: err instanceof Error ? err.message : "SMTP verification failed",
   267→    };
   268→  }
   269→}

   271→/**
   272→ * Verify the Brevo API key works by calling the account endpoint.
   273→ * This is the preferred sending path — if the API key is invalid, emails
   274→ * will silently fall back to SMTP (which may have deliverability issues
   275→ * with Microsoft 365 domains).
   276→ *
   277→ * Returns ok=true if the API key is valid AND the account can send emails.
   278→ */
   279→export async function verifyBrevoApi(): Promise<{
   280→  ok: boolean;
   281→  message: string;
   282→  usingEnvVar: string;
   283→}> {
   284→  const apiKey = process.env.BREVO_API_KEY || process.env.SMTP_PASS;
   285→  const usingEnvVar = process.env.BREVO_API_KEY ? "BREVO_API_KEY" : "SMTP_PASS (fallback — may not work as API key)";
   286→
   287→  if (!apiKey) {
   288→    return {
   289→      ok: false,
   290→      message: "No API key found. Set BREVO_API_KEY env var (generate in Brevo dashboard → SMTP & API → API Keys).",
   291→      usingEnvVar,
   292→    };
   293→  }
   294→
   295→  try {
   296→    // Call the account endpoint to verify the API key without sending an email.
   297→    // GET https://api.brevo.com/v3/account returns account info if the key is valid.
   298→    const resp = await fetch("https://api.brevo.com/v3/account", {
   299→      method: "GET",
   300→      headers: {
   301→        "accept": "application/json",
   302→        "api-key": apiKey,
   303→      },
   304→    });
   305→
   306→    if (resp.ok) {
   307→      const data = await resp.json().catch(() => ({}));
   308→      const plan = data.plan?.find((p: { type: string }) => p.type === "free") ? "free" : "paid";
   309→      return {
   310→        ok: true,
   311→        message: `Brevo API key valid. Account: ${data.email || "unknown"} (${data.companyName || "no company"}), plan: ${plan}. Emails will be sent via the API (preferred path).`,
   312→        usingEnvVar,
   313→      };
   314→    } else {
   315→      const text = await resp.text().catch(() => "");
   316→      return {
   317→        ok: false,
   318→        message: `Brevo API returned HTTP ${resp.status}. ${text.slice(0, 200)}. The API key may be invalid or this is the SMTP password (not an API key). Generate a separate API key in Brevo dashboard → SMTP & API → API Keys, and add it as BREVO_API_KEY in Vercel. Without a valid API key, emails fall back to SMTP which has deliverability issues with Microsoft 365.`,
   319→        usingEnvVar,
   320→      };
   321→    }
   322→  } catch (err) {
   323→    return {
   324→      ok: false,
   325→      message: `API verification request failed: ${err instanceof Error ? err.message : "unknown error"}. Check network connectivity to api.brevo.com.`,
   326→      usingEnvVar,
   327→    };
   328→  }
   329→}

   333→export type DnsRecord = {
   334→  name: string;     // DNS record name queried
   335→  found: boolean;
   336→  value: string | null;
   337→  explanation: string;
   338→  recommendation?: string;
   339→};
   340→
   341→export type DnsReport = {
   342→  domain: string;
   343→  spf: DnsRecord;
   344→  // Brevo's modern domain authentication uses CNAME records at
   345→  // brevo1._domainkey and brevo2._domainkey pointing to
   346→  // b1.<domain>.dkim.brevo.com and b2.<domain>.dkim.brevo.com.
   347→  // Older setups used a TXT record at brevo._domainkey. We check both.
   348→  dkimBrevo1Cname: DnsRecord;  // brevo1._domainkey CNAME → b1.<domain>.dkim.brevo.com
   349→  dkimBrevo2Cname: DnsRecord;  // brevo2._domainkey CNAME → b2.<domain>.dkim.brevo.com
   350→  dkimBrevoTxt: DnsRecord;     // brevo._domainkey TXT (legacy)
   351→  dkimS1: DnsRecord;           // s1._domainkey TXT (very old alternate)
   352→  dkimS2: DnsRecord;           // s2._domainkey TXT (very old alternate)
   353→  dmarc: DnsRecord;
   354→  mx: { exchange: string; priority: number }[];
   355→};
   356→
   357→async function resolveTxtJoined(name: string): Promise<string | null> {
   358→  try {
   359→    const records = await resolveTxt(name);
   360→    if (records.length === 0) return null;
   361→    // TXT records come back as arrays of string fragments — join them.
   362→    return records.map((r) => r.join("")).join(" ");
   363→  } catch {
   364→    return null;
   365→  }
   366→}
   367→
   368→async function resolveCnameString(name: string): Promise<string | null> {
   369→  try {
   370→    const record = await resolveCname(name);
   371→    if (record.length === 0) return null;
   372→    // resolveCname returns the canonical name without trailing dot —
   373→    // add it back for display consistency.
   374→    return record[0];
   375→  } catch {
   376→    return null;
   377→  }
   378→}

   380→/**
   381→ * Inspect the DNS records that govern email deliverability for the given
   382→ * domain. Used by the admin diagnostic tool to show the user exactly what
   383→ * is missing and provide copy-paste DNS record templates.
   384→ *
   385→ * For Brevo specifically (modern domain authentication):
   386→ *   SPF:        v=spf1 include:spf.brevo.com ~all  (TXT on @)
   387→ *   DKIM:       brevo1._domainkey CNAME → b1.<domain>.dkim.brevo.com
   388→ *               brevo2._domainkey CNAME → b2.<domain>.dkim.brevo.com
   389→ *   DMARC:      v=DMARC1; p=none; rua=mailto:...  (TXT on _dmarc)
   390→ *   Brevo code: brevo-code:<hash>  (TXT on @, used by Brevo to verify ownership)
   391→ *
   392→ * IMPORTANT: SPF must be a SINGLE TXT record with all `include:` directives
   393→ * inside it. Multiple SPF TXT records on the same name is invalid per RFC 7208
   394→ * and causes recipient servers to ignore SPF entirely.
   395→ */
   396→export async function checkEmailDnsRecords(
   397→  domain: string = "phronesis-studio.com"
   398→): Promise<DnsReport> {
   399→  const bareDomain = domain.replace(/^https?:\/\//, "").replace(/\/$/, "");
   400→
   401→  const [
   402→    rootTxtRecords,
   403→    dkimBrevo1Cname,
   404→    dkimBrevo2Cname,
   405→    dkimBrevoTxt,
   406→    dkimS1,
   407→    dkimS2,
   408→    dmarc,
   409→    mxRecords,
   410→  ] = await Promise.all([
   411→    resolveTxtJoined(bareDomain),
   412→    resolveCnameString(`brevo1._domainkey.${bareDomain}`),
   413→    resolveCnameString(`brevo2._domainkey.${bareDomain}`),
   414→    resolveTxtJoined(`brevo._domainkey.${bareDomain}`),
   415→    resolveTxtJoined(`s1._domainkey.${bareDomain}`),
   416→    resolveTxtJoined(`s2._domainkey.${bareDomain}`),
   417→    resolveTxtJoined(`_dmarc.${bareDomain}`),
   418→    resolveMx(bareDomain).catch(() => []),
   419→  ]);
   420→
   421→  // rootTxtRecords may contain MULTIPLE TXT records (SPF + brevo-code + …).
   422→  // Find the SPF one specifically.
   423→  const spfRecord = rootTxtRecords && /v=spf1/.test(rootTxtRecords)
   424→    ? rootTxtRecords.split(" ").find(r => /v=spf1/.test(r)) || rootTxtRecords
   425→    : null;
   426→  // Actually: when resolveTxt returns multiple TXT records, our join
   427→  // concatenates them with spaces, which is wrong. We need to find which
   428→  // one starts with v=spf1. Let's be smarter:
   429→  let spfValue: string | null = null;
   430→  if (rootTxtRecords) {
   431→    // Each TXT record came back as a joined string; we joined those with
   432→    // " ". To find SPF, we look for the v=spf1 token anywhere.
   433→    const spfMatch = rootTxtRecords.match(/v=spf1[^"\n]*/i);
   434→    spfValue = spfMatch ? spfMatch[0] : null;
   435→  }
   436→  // Also: detect if there's a brevo-code TXT (separate from SPF).
   437→  const hasBrevoCode = rootTxtRecords && /brevo-code:/i.test(rootTxtRecords);
   438→  // Check that SPF includes spf.brevo.com
   439→  const spfIncludesBrevo = Boolean(spfValue && /include:spf\.brevo\.com/i.test(spfValue));
   440→
   441→  // Determine if any DKIM is configured (CNAME form preferred, TXT form legacy)
   442→  const dkimCnameFound = Boolean(dkimBrevo1Cname || dkimBrevo2Cname);
   443→  const dkimTxtFound = Boolean(dkimBrevoTxt && /v=DKIM1/.test(dkimBrevoTxt));
   444→
   445→  return {
   446→    domain: bareDomain,
   447→    spf: {
   448→      name: `${bareDomain}  TXT`,
   449→      found: Boolean(spfValue),
   450→      value: spfValue,
   451→      explanation:
   452→        "SPF (Sender Policy Framework) lists which mail servers are allowed to send email from your domain. Without SPF including Brevo, Gmail/Outlook will silently filter most of your email to spam. IMPORTANT: SPF must be a SINGLE TXT record with all include: directives inside it — multiple SPF TXT records on the same name is invalid and causes SPF to be ignored entirely.",
   453→      recommendation: !spfValue
   454→        ? `Add this TXT record at your DNS provider:
   455→  Name: @  (or ${bareDomain})
   456→  Value: v=spf1 include:spf.brevo.com ~all`
   457→        : spfValue && !spfIncludesBrevo
   458→          ? `Your SPF record exists but does NOT include Brevo. EDIT the existing TXT record (do not add a second SPF record) to:
   459→  v=spf1 include:spf.brevo.com ${spfValue.replace(/^v=spf1\s*/, "")}
   460→
   461→Current: ${spfValue}
   462→
   463→Example (if you also have Namecheap email forwarding):
   464→  v=spf1 include:spf.brevo.com include:spf.efwd.registrar-servers.com ~all`
   465→          : undefined,
   466→    },
   467→    dkimBrevo1Cname: {
   468→      name: `brevo1._domainkey.${bareDomain}  CNAME`,
   469→      found: Boolean(dkimBrevo1Cname),
   470→      value: dkimBrevo1Cname ? `${dkimBrevo1Cname}.` : null,
   471→      explanation:
   472→        "DKIM selector #1 — Brevo's modern domain authentication uses a CNAME record pointing to b1.<domain>.dkim.brevo.com. This is the recommended setup; Brevo generates it in their dashboard under Senders & IP → Authenticate your domain.",
   473→      recommendation: !dkimBrevo1Cname && !dkimTxtFound
   474→        ? `Log in to Brevo dashboard → Senders & IP → Authenticate your domain.
   475→Brevo will give you a CNAME record to add at:
   476→  Name: brevo1._domainkey
   477→  Value: b1.${bareDomain}.dkim.brevo.com.`
   478→        : undefined,
   479→    },
   480→    dkimBrevo2Cname: {
   481→      name: `brevo2._domainkey.${bareDomain}  CNAME`,
   482→      found: Boolean(dkimBrevo2Cname),
   483→      value: dkimBrevo2Cname ? `${dkimBrevo2Cname}.` : null,
   484→      explanation:
   485→        "DKIM selector #2 — Brevo's modern domain authentication uses a CNAME record pointing to b2.<domain>.dkim.brevo.com. This is the recommended setup; Brevo generates it in their dashboard under Senders & IP → Authenticate your domain.",
   486→      recommendation: !dkimBrevo2Cname && !dkimTxtFound
   487→        ? `Log in to Brevo dashboard → Senders & IP → Authenticate your domain.
   488→Brevo will give you a CNAME record to add at:
   489→  Name: brevo2._domainkey
   490→  Value: b2.${bareDomain}.dkim.brevo.com.`
   491→        : undefined,
   492→    },
   493→    dkimBrevoTxt: {
   494→      name: `brevo._domainkey.${bareDomain}  TXT`,
   495→      found: dkimTxtFound,
   496→      value: dkimBrevoTxt,
   497→      explanation:
   498→        "Legacy DKIM TXT record (older Brevo setup). Most modern Brevo configurations use the CNAME form above instead; either is acceptable.",
   499→    },
   500→    dkimS1: {
   501→      name: `s1._domainkey.${bareDomain}  TXT`,
   502→      found: Boolean(dkimS1 && /v=DKIM1/.test(dkimS1)),
   503→      value: dkimS1,
   504→      explanation: "Alternate DKIM selector some older Brevo setups use.",
   505→    },
   506→    dkimS2: {

   953→/**
   954→ * Sends an email to ALL subscribers announcing new content (episode, article,
   955→ * or guide). This closes Bug 2: previously no code sent notifications when
   956→ * new content was published.
   957→ *
   958→ * Rate-limited (5 emails/sec) to respect Brevo free-tier limits (~300/day).
   959→ * Writes one SentEmail audit row per recipient. Non-fatal on individual
   960→ * failures — returns a summary of sent/failed/rejected.
   961→ *
   962→ * @returns { sent, failed, rejected } counts
   963→ */
   964→export async function sendPublicationBroadcast(opts: {
   965→  subject: string;
   966→  html: string;
   967→  text: string;
   968→  locale?: "en" | "ar" | "all"; // filter by subscriber locale, or "all"
   969→}): Promise<{ sent: number; failed: number; rejected: string[] }> {
   970→  const { subject, html, text, locale = "all" } = opts;
   971→
   972→  if (!isEmailConfigured()) {
   973→    console.warn("[broadcast] Email not configured — skipping broadcast.");
   974→    return { sent: 0, failed: 0, rejected: [] };
   975→  }
   976→
   977→  // Fetch subscribers
   978→  const { db } = await import("@/lib/db");
   979→  const where = locale === "all" ? {} : { locale };
   980→  const subscribers = await db.subscriber.findMany({
   981→    where,
   982→    select: { email: true, locale: true },
   983→  });
   984→
   985→  if (subscribers.length === 0) {
   986→    return { sent: 0, failed: 0, rejected: [] };
   987→  }
   988→
   989→  let sent = 0;
   990→  let failed = 0;
   991→  const rejected: string[] = [];
   992→  const replyTo = process.env.CONTACT_EMAIL || "ahmed@phronesis-studio.com";
   993→
   994→  for (const sub of subscribers) {
   995→    try {
   996→      const result = await sendEmail({
   997→        to: sub.email,
   998→        subject,
   999→        html,
  1000→        text,
  1001→        replyTo,
  1002→      });
  1003→
  1004→      if (result.rejected.includes(sub.email)) {
  1005→        rejected.push(sub.email);
  1006→        failed++;
  1007→      } else {
  1008→        sent++;
  1009→      }
  1010→
  1011→      // Write audit log
  1012→      try {
  1013→        await db.sentEmail.create({
  1014→          data: {
  1015→            toEmail: sub.email,
  1016→            subject,
  1017→            bodyText: text,
  1018→            bodyHtml: html,
  1019→          },
  1020→        });
  1021→      } catch (logErr) {
  1022→        console.error("[broadcast] SentEmail log failed:", logErr);
  1023→      }
  1024→
  1025→      // Rate limit: 5 emails/sec (200ms between sends)
  1026→      await new Promise((r) => setTimeout(r, 200));
  1027→    } catch (err) {
  1028→      console.error(`[broadcast] Failed to send to ${sub.email}:`, err);
  1029→      failed++;
  1030→      rejected.push(sub.email);
  1031→    }
  1032→  }
  1033→
  1034→  console.log(
  1035→    `[broadcast] "${subject}" — sent: ${sent}, failed: ${failed}, rejected: ${rejected.length}`
  1036→  );
  1037→
  1038→  return { sent, failed, rejected };
  1039→}
